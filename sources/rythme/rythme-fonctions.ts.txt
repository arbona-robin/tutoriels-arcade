// Notions : fonctions à paramètre (et avec retour pour creerCible), boucle sur tous les sprites d'un type,
// chevauchement testé à l'instant de l'appui, événement « est détruit ».

    export const Cible = SpriteKind.create()

function creerNote (ligne: number) {
    note = sprites.create(assets.image`note`, SpriteKind.Note)
    note.x = 165
    note.y = ligne * 20 - 10
    note.vx = -50
    note.setFlag(SpriteFlag.AutoDestroy, true)
}
function frapper (cibleFrappee: Sprite) {
    info.changeScoreBy(-1)
    for (let uneNote of sprites.allOfKind(SpriteKind.Note)) {
        if (uneNote.overlapsWith(cibleFrappee)) {
            info.changeScoreBy(2)
            sprites.destroy(uneNote, effects.hearts, 300)
        }
    }
}
controller.up.onEvent(ControllerButtonEvent.Pressed, function () {
    music.play(music.tonePlayable(262, music.beat(BeatFraction.Half)), music.PlaybackMode.InBackground)
    frapper(cibleHaut)
})
controller.A.onEvent(ControllerButtonEvent.Pressed, function () {
    music.play(music.tonePlayable(294, music.beat(BeatFraction.Half)), music.PlaybackMode.InBackground)
    frapper(cibleMilieu)
})
controller.down.onEvent(ControllerButtonEvent.Pressed, function () {
    music.play(music.tonePlayable(330, music.beat(BeatFraction.Half)), music.PlaybackMode.InBackground)
    frapper(cibleBas)
})
sprites.onDestroyed(SpriteKind.Note, function (sprite) {
    if (sprite.x < 0) {
        info.changeScoreBy(-1)
    }
})
let note: Sprite = null
let cible: Sprite = null
let cibleHaut: Sprite = null
let cibleMilieu: Sprite = null
let cibleBas: Sprite = null
info.setScore(0)
scene.setBackgroundImage(assets.image`partition`)
cibleHaut = creerCible(1)
cibleMilieu = creerCible(2)
cibleBas = creerCible(3)
